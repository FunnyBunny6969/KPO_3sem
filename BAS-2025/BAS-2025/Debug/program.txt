// Функция демонстрации переполнения
uint function demonstrate_underflow()
{
    print 'демонстрации переполнения';

    uint a; uint b; uint c;
    
    a = 0x05;   print 'a = '; print a;
    b = 0xa;    print 'b = '; print b;
    
    // Выражение, результат которого "меньше нуля" для uint
    c = a - b; // Будет 4294967291
    print 'c = '; print a - b;
    
    return 0+5;
};


// Функция с вызовом стандартной библиотеки
uint function use_standard_library()
{    
    print 'Вызов библиотечных функций';
    
    print 'pow(uint, uint)';
    print pow(2, 2);
    print pow(3, 3);

    print 'random(uint min, uint max)';
    print random(0, 100);
    print random(1, 9);

    return 0;
};


// Функция с побитовыми операциями и switch
uint function bitwise_ops(uint result, uint op_type)
{        
    // Switch для выбора битовой операции
    switch (op_type)
    {
    case 0x00: // операция И (AND) с 0x0F
        result = result & 0x0F;
        break;
        
    case 0x01: // операция ИЛИ (OR) с 0xF0
        result = result | 0xf0;
        break;
        
    case 0x02: // операция инверсия (NOT)
        result = ~result;
        break;
        
    default: // умножить на 2
        result = result * 2;

        switch(result){
        case 200: result = 95; break;
        default: result = 55;
        };

        break;
    };
    return result;
};


// Главная функция
main
{
    uint test_uint; string test_string; char test_char;

    // Проверка значений до инициализации
    print 'ДО';
    print 'test_uint = '; print test_uint;
    print 'test_string = '; print test_string;
    print 'test_char = '; print test_char;
    
    // Инициализация
    test_uint = 5;
    test_string = 'ТЕСТОВАЯ СТРОКА';
    test_char = '@';

    // Проверка значений ПОСЛЕ инициализации
    print 'ПОСЛЕ';
    print 'test_uint = '; print test_uint;
    print 'test_string = '; print test_string;
    print 'test_char = '; print test_char;


    // Демонстрация переполнения
    demonstrate_underflow();
    

    // Демонстрация вызовов стандартной библиотеки
    use_standard_library();
    

    print '=== ТЕСТ БИТОВЫХ ОПЕРАЦИЙ ===';
    
    // 4 вызова с заготовленными значениями
    print 'bitwise_ops(0x12, 0x00) = ';
    print bitwise_ops(0x12, 0x00); // 0x12 & 0x0F = 0x02
    
    print 'bitwise_ops(0x12, 0x01) = ';
    print bitwise_ops(0x12, 0x01); // 0x12 | 0xF0 = 0xF2 (десятичное 242)
    
    print 'bitwise_ops(0x12, 0x02) = ';
    print bitwise_ops(0x12, 0x02); // ~0x12 = 0xFFFFFFED (десятичное 4294967285)
    
    print 'bitwise_ops(0x12, 0x03) = ';
    print bitwise_ops(0x12, 0x03); // 0x12 * 2 = 0x24 (десятичное 36)

    
    return 0;
};
