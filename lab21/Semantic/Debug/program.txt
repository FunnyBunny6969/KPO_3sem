// Цель: Проверка прав доступа по битовой маске.
unsigned_integer function check_permissions(unsigned_integer user_role, unsigned_integer required_mask)
{
    declare unsigned_integer access_level;
	
    // Побитовое ИЛИ: Объединяем роль пользователя и требуемую маску (фиктивная операция)
    access_level = user_role | required_mask; 

    // Побитовое И: Проверяем младший байт (0xFF). 
    // Используем switch для принятия решения на основе результата битовой операции.
    switch (access_level & 0x07) // Проверяем 3 младших бита
    {
        case 0x01: // Десятичное 1
            access_level = access_level + 10;
            break;
        case 0x02: // Десятичное 2
            access_level = access_level + 20;
            break;
        case 0x03:
        case 0x3:
        case 0x55:
            access_level = access_level + 30;
            break;
        default:
            // Побитовая инверсия: ~
            access_level = ~access_level; 
            break;
    };
    
    return access_level;
};

// --- 2. Функция для работы со строками, char и стандартными функциями
// Цель: Создать уникальный идентификатор, объединив часть строки и символ.
string function create_id(string name_part, char delimiter)
{
    // Локальное объявление стандартной функции для подстроки (для SA)
    declare string function substr(string a, unsigned_integer p, unsigned_integer n);
    
    declare unsigned_integer initial_len;
    
    // Вызов стандартной функции (strlen - предполагаем, что она существует)
    initial_len = strlen(name_part); 
    
    declare unsigned_integer random_seed;
    // Вызов стандартной функции: возведение в степень (фиктивно, для проверки power)
    random_seed = power(3, initial_len); 

    // Возвращаем первые 4 символа (substr) и добавляем разделитель (char)
    // СА проверяет: (string) + (char) = (string)
    return substr(name_part, 0x01, 4) + delimiter; 
};
