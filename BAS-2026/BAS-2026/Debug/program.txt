// 2. Функция с параметрами: побитовые операции (&&, ||, ~) по заданию
short function test_bitwise(short a, short b)
{
    short res_and;
    short res_or;
    short res_not;

    res_and = a && 0x0F;
    res_or  = b || 0xF0;
    res_not = ~a;

    print '--- Побитовые операции ---';
    print res_and;
    print res_or;
    print res_not;

    return (res_and || res_or) && ~0x00;
};

// 3. Вложенные циклы while (while внутри while) и унарный минус в условиях
short function test_nested_while(short limit)
{
    short i;
    short j;
    short total;

    i = -2;
    total = 0;

    // Внешний цикл с отрицательным стартом и сравнением
    while (i < limit)
    {
        j = 2;
        // Внутренний цикл с унарным минусом в условии
        while (j > -1)
        {
            total = total + 1;
            j = j - 1;
        };
        i = i + 1;
    };

    return total;
};
